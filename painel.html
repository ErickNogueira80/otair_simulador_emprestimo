function doPost(e) { try { var dados = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var abaEmprestimos = ss.getSheetByName("Emprestimos"); var abaParcelas = ss.getSheetByName("Parcelas"); var dataAtual = new Date(); var idEmprestimo = "EMP_" + Math.floor(Math.random() * 100000); abaEmprestimos.appendRow([ idEmprestimo, dados.nomeCliente, dados.valorEmprestado, dados.taxaJuros, dados.qtdParcelas, dados.valorTotal, Utilities.formatDate(dataAtual, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss") ]); var valorParcela = dados.valorTotal / dados.qtdParcelas; for (var i = 1; i <= dados.qtdParcelas; i++) { var dataVencimento = new Date(); dataVencimento.setDate(dataVencimento.getDate() + i); // Vencimento diário (+1 dia por parcela) abaParcelas.appendRow([ idEmprestimo, i + "º Dia", valorParcela.toFixed(2), Utilities.formatDate(dataVencimento, Session.getScriptTimeZone(), "dd/MM/yyyy"), "Pendente" ]); } return ContentService.createTextOutput(JSON.stringify({ "status": "sucesso", "mensagem": "Empréstimo registrado com sucesso!", "id": idEmprestimo })).setMimeType(ContentService.MimeType.JSON); } catch (erro) { return ContentService.createTextOutput(JSON.stringify({ "status": "erro", "mensagem": erro.toString() })).setMimeType(ContentService.MimeType.JSON); } }
